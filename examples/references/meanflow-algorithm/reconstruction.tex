\texttt{\cm{\# net: x-prediction network}}\\
\texttt{\cm{\# x: training batch in pixels}}\\[0.7em]
\texttt{t, r = \fn{sample\_t\_r}()}\\
\texttt{e = \fn{randn\_like}(x)}\\[0.7em]
\texttt{z = (1 \op{-} t) \op{*} x \op{+} t \op{*} e}\\[0.7em]
\texttt{\cm{\# average velocity u from x-prediction}}\\
\texttt{\kw{def} u\_fn(z, r, t):}\\
\texttt{\hspace*{2em}\kw{return} (z \op{-} net(z, r, t)) \op{/} t}\\[0.7em]
\texttt{\cm{\# instantaneous velocity v at time t}}\\
\texttt{v = u\_fn(z, t, t)}\\[0.7em]
\texttt{\cm{\# predict u and dudt}}\\
\texttt{u, dudt = \fn{jvp}(u\_fn, (z, r, t), (v, 0, 1))}\\[0.7em]
\texttt{\cm{\# compound function V}}\\
\texttt{V = u \op{+} (t \op{-} r) \op{*} \kw{stopgrad}(dudt)}\\[0.7em]
\texttt{loss = \fn{metric}(V, e \op{-} x)}\\
